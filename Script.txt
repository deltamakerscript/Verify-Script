local realName = LocalPlayer.Name
local realDisplayName = LocalPlayer.DisplayName

local function enforceNameOnLabel(label)
	if not label:IsA("TextLabel") then return end

	local function checkAndReplace()
		local text = label.Text
		local targetDisplay = customDisplayName .. VERIFIED_SYMBOL
		local targetUser = "@" .. customUsername

		-- Username (@)
		if string.find(text, "@" .. realName) and text ~= targetUser then
			label.Text = targetUser
		-- Display Name
		elseif (text == realDisplayName or string.find(text, realDisplayName)) and text ~= targetDisplay then
			label.Text = targetDisplay
		end
	end

	checkAndReplace()

	if not label:GetAttribute("InstantNameHooked") then
		label:SetAttribute("InstantNameHooked", true)
		label:GetPropertyChangedSignal("Text"):Connect(checkAndReplace)
	end
end

local function hookGuiContainer(container)
	if not container then return end
	
	for _, desc in ipairs(container:GetDescendants()) do
		enforceNameOnLabel(desc)
	end
	
	container.DescendantAdded:Connect(function(desc)
		enforceNameOnLabel(desc)
	end)
end

-- Send Fake Message
local function sendMessage()
	local message = messageBox.Text
	local senderName = isLocked and customChatSender or (senderNameBox.Text ~= "" and senderNameBox.Text or customDisplayName)
	local badgeText = isChatBadgeEnabled and VERIFIED_SYMBOL or ""

	if message ~= "" then
		pcall(function()
			if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
				local generalChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
				if generalChannel then
					generalChannel:DisplaySystemMessage("<font color='" .. selectedColorHex .. "'>" .. senderName .. badgeText .. ":</font> " .. message)
				end
			else
				game:GetService("StarterGui"):SetCore("ChatMakeSystemMessage", {
					Text = senderName .. badgeText .. ": " .. message,
					Color = selectedColorRGB,
					Font = Enum.Font.SourceSansBold,
					TextSize = 18
				})
			end
			
			-- Bubble Chat
			local character = LocalPlayer.Character
			if character and character:FindFirstChild("Head") then
				if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
					TextChatService:DisplayBubble(character.Head, message)
				else
					ChatService:Chat(character.Head, message, Enum.ChatColor.White)
				end
			end
		end)

		messageBox.Text = ""
	end
end

sendButton.MouseButton1Click:Connect(sendMessage)
messageBox.FocusLost:Connect(function(enterPressed)
	if enterPressed then sendMessage() end
end)
